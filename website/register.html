// register submit const form = document.getElementById("regForm"); const btn = document.getElementById("registerBtn"); form.addEventListener("submit", async (e) => { e.preventDefault(); const username = document.getElementById("username").value.trim(); const email = document.getElementById("email").value.trim(); const grade = document.getElementById("grade").value; const password = document.getElementById("password").value; const confirmPassword = document.getElementById("confirmPassword").value; if(!username || !email || !grade || !password || !confirmPassword){ setMsg("Please fill in all fields.", "error"); return; } if(!validEmail(email)){ setMsg("Please enter a valid email address.", "error"); return; } if(password.length < 6){ setMsg("Password must be at least 6 characters.", "error"); return; } if(password !== confirmPassword){ setMsg("Passwords do not match.", "error"); return; } btn.disabled = true; setMsg("Creating account..."); try { const res = await fetch(`${API}/api/auth/register`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ username, email, grade, password }) }); const data = await res.json().catch(()=> ({})); if(res.ok){ setMsg( "Account created ✅ Please check your email to verify your account before logging in.", "ok" ); // Optional redirect to info page setTimeout(() => { window.location.href = "verify-info.html"; }, 1500); } else { setMsg(data.message || "Could not register.", "error"); btn.disabled = false; } } catch (err) { setMsg("Network error. Please try again.", "error"); btn.disabled = false; } });
